cmake_minimum_required(VERSION 3.24)
project(Aero7CredentialVault VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(KF6Wallet REQUIRED)
find_package(Aero7Qt REQUIRED)
add_library(vault-ui STATIC src/VaultWindow.cpp src/VaultBackend.h)
target_include_directories(vault-ui PUBLIC src)
target_link_libraries(vault-ui PUBLIC Qt6::Widgets)
add_executable(aero7-credential-vault src/main.cpp src/KWalletBackend.cpp)
qt_add_resources(aero7-credential-vault icons PREFIX /aero7 FILES icons/credential-vault.png)
target_link_libraries(aero7-credential-vault PRIVATE vault-ui Qt6::DBus KF6::Wallet Aero7Qt::Aero7Qt)
install(TARGETS aero7-credential-vault RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/aero7-credential-vault.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES icons/credential-vault.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps RENAME aero7-credential-vault.png)
# These defaults belong only to the optional package. User overrides still win.
install(FILES data/kwalletrc DESTINATION /etc/xdg/aero7-desktop)
install(FILES data/aero7-portals.conf DESTINATION /etc/xdg/aero7-desktop/xdg-desktop-portal)
install(FILES README.md icons/PROVENANCE.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/aero7-credential-vault)
install(FILES icons/UPSTREAM-LICENSE icons/UPSTREAM-README.md
    DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/aero7-credential-vault)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(vault-ui-test tests/VaultWindowTest.cpp)
    target_link_libraries(vault-ui-test PRIVATE vault-ui Qt6::Test)
    add_test(NAME vault-ui COMMAND vault-ui-test)
    set_tests_properties(vault-ui PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen)
    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
    add_executable(vault-backend-test tests/KWalletBackendTest.cpp src/KWalletBackend.cpp)
    target_link_libraries(vault-backend-test PRIVATE vault-ui Qt6::DBus Qt6::Test KF6::Wallet)
    add_test(NAME vault-backend COMMAND ${DBUS_RUN_SESSION}
        --config-file=${CMAKE_CURRENT_SOURCE_DIR}/tests/dbus-test.conf -- $<TARGET_FILE:vault-backend-test>)
    set_tests_properties(vault-backend PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen TIMEOUT 45)
endif()
